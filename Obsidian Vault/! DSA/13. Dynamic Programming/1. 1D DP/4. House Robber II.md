## Space Optimized
```cpp
class Solution {
public:
    int helper(vector<int> &nums){
        int n = nums.size();
        if(n == 0) return 0;
        if(n == 1) return nums[0];
        if(n == 2) return max(nums[0], nums[1]);
  
        int prev2 = nums[0];
        int prev = max(nums[0], nums[1]);
        int cur_i = 0;
        for(int i = 2; i < n; i++){
            int rob = prev2 + nums[i];
            int notrob = prev;
            cur_i = max(rob, notrob);
  
            prev2 = prev;
            prev = cur_i;
        }
  
        return cur_i;
    }

    int rob(vector<int>& nums) {
        int n = nums.size();
  
        if(n == 0) return 0;
        if(n == 1) return nums[0];
        if(n == 2) return max(nums[0], nums[1]);
        vector<int> one;
        vector<int> two;
        for(int i = 0; i < n; i++){
            if(i != 0) one.push_back(nums[i]);
            if(i != n-1) two.push_back(nums[i]);
        }
        int onemax = helper(one);
        int twomax = helper(two);
        return max(onemax, twomax);
    }
};
```
