class Solution {
public:
    int helper(vector<int> &money){
        int n = money.size();
        int prev = money[0];
        int prev2 = 0;

        for(int i = 1; i < n; i++){
            int pick = money[i];
            if(i > 1) pick += prev2;

            int notpick = prev;

            int cur_i = max(pick, notpick);

            prev2 = prev;
            prev = cur_i;
        }
        return prev;
    }

    int houseRobber(vector<int>& money) {
        int n = money.size();
        vector<int> temp;
        vector<int> temp2;

        if(n == 1) return money[0];

        for(int i = 0; i < n; i++){
            if(i != 0) temp.push_back(money[i]);
            if(i != n - 1) temp2.push_back(money[i]);
        }

        return max(helper(temp), helper(temp2));
    }
};
