/**
 * Definition for a binary tree node.
 * class TreeNode {
 * public:
 *     int data;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int val) : data(val), left(nullptr), right(nullptr) {}
 * };
 **/

class Solution {
private:
    int floor(TreeNode* root, int key) {
        int ans = -1;
        TreeNode* curr = root;
        while (curr) {
            if (curr->data == key) {
                return key;
            } else if (curr->data > key) {
                curr = curr->left;
            } else {
                ans = curr->data; // potential floor
                curr = curr->right;
            }
        }
        return ans;
    }

    int ceil(TreeNode* root, int key) {
        int ans = -1;
        TreeNode* curr = root;
        while (curr) {
            if (curr->data == key) {
                return key;
            } else if (curr->data < key) {
                curr = curr->right;
            } else {
                ans = curr->data; // potential ceil
                curr = curr->left;
            }
        }
        return ans;
    }

public:
    vector<int> floorCeilOfBST(TreeNode* root, int key) {
        return {floor(root, key), ceil(root, key)};
    }
};
