

### Recursion
1. we first set an index at the first position
2. we then use i to iterate from index to end of string
3. swap the index with i and then recursion using index + 1
4. that will go through the string and attempt to swap the elements 
	   like ABC, ACB 
		firstly it will swap A and A giving ABC
		then B and B giving ABC
		then C and C giving ABC
		then since it reached the end, it will backtrack and i will get incremented, causing index at 1 to swap with i at 2 forming ACB
		then that will too backtrack and swap index at 0 with i at 1 causing BAC, which further goes on to swap A and C giving BCA 
		then further backtracking we get index at 0 and i at 2 causing CBA and further getting CAB
5. thus all 3! = 6 permutations get formed and pushed into the vector
6. also before each for loop, we check if the character is already present in a set because if it is it just gives us duplicate strings and we will thus skip that iteration


```cpp
class Solution {
    private:
    void recursivePermute(int index, string &s, vector<string> &ans){
        if(index == s.size()){
            ans.push_back(s);
            return;
        }
        
        unordered_set<char> used;
        
        for(int i = index; i < s.size(); i++){
            if(used.find(s[i]) != used.end()) continue;
            used.insert(s[i]);
            
            swap(s[index], s[i]);
            recursivePermute(index + 1, s, ans);
            swap(s[index], s[i]);
        }
    }
  public:
    vector<string> findPermutation(string &s) {
        vector<string> ans;
        recursivePermute(0, s, ans);
        return ans;
    }
};
```
