- **Moore's Voting Algorithm**

```cpp
class Solution {
public:
    int majorityElement(vector<int>& nums) {
        int element;
        int count = 0;

        for(int i = 0; i < nums.size(); i++){
            if(count == 0){
                count = 1;
                element = nums[i];
            } else if(element == nums[i]){
                count++;
            } else {
                count--;
            }
        }

        return element;
    }
};
```
1. here we first select an element and increment the count for every time it appears,
2. if it goes to zero that means it cannot dominate over the other elements, so we select another element
