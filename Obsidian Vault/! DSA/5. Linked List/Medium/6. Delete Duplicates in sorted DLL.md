/*
Definition of doubly linked list:
struct ListNode
{
    int val;
    ListNode *next;
    ListNode *prev;
    ListNode()
    {
        val = 0;
        next = NULL;
        prev = NULL;
    }
    ListNode(int data1)
    {
        val = data1;
        next = NULL;
        prev = NULL;
    }
    ListNode(int data1, ListNode *next1, ListNode *prev1)
    {
        val = data1;
        next = next1;
        prev = prev1;
    }
};
*/

class Solution {
public:
    ListNode * removeDuplicates(ListNode *head) {
        ListNode* temp = head;
// use temp for iterating through DLL
        while(temp != nullptr){
        // normal stuff
            ListNode* nextNode = temp->next;
// we create a nextNode that checks the nextNode .. duh
// as long as that next node is not the end of DLL (nullptr) and the values are duplicates (aka 1 1 1 for eg) it keeps going and reaches the end of duplicates
            while(nextNode != nullptr && nextNode->val == temp->val){
                ListNode* duplicate = nextNode;
                nextNode = nextNode->next;
                delete duplicate;
            }
// we keep moving the nextNode and deleting all the repeated elements
            
            temp->next = nextNode;
// set temps next to after the duplicates
            if(nextNode != nullptr){
                nextNode->prev = temp;
            }
// if nextnode is at the end aka nullptr we cant reference it or set a prev pointer thats why this condition exists

            temp = temp->next;
// move along the DLL
        }
        return head;
    }
};
