Problem Statement:  
Given a string S, the task is to find the minimum flips required to convert an initial binary string consisting of only zeroes to S where every flip of a character flips all succeeding characters as well.

Input:  
S = “01011”  
Output:  
3

Explanation:  
Initial String – 00000  
Flip the 2nd bit – 01111  
Flip the 3rd bit – 01000  
Flip the 4th bit – 01011  
Total Flips = 3

Input:  
S = “01001”  
Output:  
3

Explanation:  
Initial String – 00000  
Flip the 2nd bit – 01111  
Flip the 3rd bit – 01000  
Flip the 5th bit – 01001  
Total Flips = 3

## My Approach
```cpp
#include <bits/stdc++.h>

class Solution{
public:
	void flip(string &temp, int index){
		for(int i = index; i < temp.size(); i++){
			temp[i] = (temp[i] == '0') ? '1' : '0';
		}
	}

	int function(string s){
		string stockk = "";
		int ans = 0;
		for(int i = 0; i < s.size(); i++) stockk.push_back('0');
		for(int i = 0; i < s.size(); i++){
			if(s[i] != stockk[i]){
				flip(stockk, i);
				ans++;
			}
		}
		return ans;
	}
};

int main(){
	string input;
	cout << "Enter a string";
	cin >> input;
	Solution sol;
	cout << sol.function(input) << endl;
	return 0;
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    int function(string s) {
        int ans = 0;
        int flip_count = 0;
        for (int i = 0; i < s.size(); i++) {
            char current = (flip_count % 2 == 0) ? '0' : '1';
            if (s[i] != current) {
                ans++;
                flip_count++;
            }
        }
        return ans;
    }
};

int main() {
    string input;
    cout << "Enter a binary string: ";
    cin >> input;

    Solution sol;
    cout << sol.function(input) << endl;
    return 0;
}
```
