
Given an n x m grid, where each cell has the following values : 

2 - represents a rotten orange
1 - represents a Fresh orange
0 - represents an Empty Cell

Every minute, if a fresh orange is adjacent to a rotten orange in **4-direction** ( upward, downwards, right, and left ) it becomes rotten. 
  
Return the **minimum** number of **minutes** required such that none of the cells has a Fresh Orange. If it's not possible, return -1.

Examples:
**Input:** grid = [ [2, 1, 1] , [0, 1, 1] , [1, 0, 1] ]
**Output:** -1
**Explanation:** Orange at (3,0) cannot be rotten.

**Input:** grid = [ [2,1,1] , [1,1,0] , [0,1,1] ] 
**Output:** 4
**Explanation:**
![](https://static.takeuforward.org/content/ProblemSetter-yuB56suz)


### Intuition
1. We use Breadth First Search so that multiple rotten oranges can infect simultaneously. Using DFS would mean following only one orange around.
2. We first note down all the rotten oranges and put them into a queue
3. Then while the queue isn't empty, we take the first rotten orange in the queue and check its neighbors for any fresh oranges
	   we place the first rotten orange into the visited vector. This vector keeps track of all the oranges that have been visited by us in BFS.
4. If we find a neighbor, we then infect that orange in that loop if it has not been visited and whether the indices (i and j) are legal and not out of bounds (ArrayOutOfBounds)
5. the time only increments after one loop has been completed, where we check whether the queue is empty and that there are still rotten oranges
	   after that loop resets, we increment time
		   this is because in the same minute multiple rotten oranges can infect others
6. 
```cpp
class Solution {
public:
// these are for checking the neighbors. For a orange at [i,j] its neighbours are i+1,j i-1,j ... so we use a for loop and iterate through this vector to simulate checking the neighbors
    vector<int> drow = {-1, 0, 1, 0};
    vector<int> dcol = {0, -1, 0, 1};

// function to prevent ArrayOutOfBounds
    bool isSafe(int i, int j, int m, int n){
        return (i >= 0 && i < m && j >= 0 && j < n);
    }

//main function
    int orangesRotting(vector<vector<int>> &grid) {
        if(grid.empty()) return 0;

        int m = grid.size();
        int n = grid[0].size();

// to note down the rotten oranges
        queue<pair<int, int>> Oranges;
        vector<vector<int>> visited(m, vector<int>(n, 0));
// initializes a vector with all 0s

        int freshOranges = 0;
        int time = 0;

        for(int i = 0; i < m; i++){
            for(int j = 0; j < n; j++){
                if(grid[i][j] == 1){
                    freshOranges++;
                    // fresh here
                } else if(grid[i][j] == 2){
                    Oranges.push({i,j});
                    visited[i][j] = 2;
                    // is rotten and since this is already rotten we add it to visited vector
                }
            }
        }

// while there is still rotten oranges
        while(!Oranges.empty()){
            int sz = Oranges.size();
            bool Orangerotten = false;
// this bool is to check whether an orange has been infected or not

            for(int k = 0; k < sz; k++){
                int row = Oranges.front().first;
                int col = Oranges.front().second;
                Oranges.pop();

// checking neighbors
                for(int i = 0; i < 4; i++){
                    int anotherRow = row + drow[i];
                    int anotherCol = col + dcol[i];


                    if(isSafe(anotherRow, anotherCol, m, n)){
// if the orange is fresh and unvisited, we make it rotten and add it to the queue and mark it as visited aswell
                        if(grid[anotherRow][anotherCol] == 1 &&
                           visited[anotherRow][anotherCol] != 2){
                            Oranges.push({anotherRow, anotherCol});
                            visited[anotherRow][anotherCol] = 2;
                            grid[anotherRow][anotherCol] = 2;
                            freshOranges--;
// decrement number of fresh oranges
                            Orangerotten = true;
// so that we can know to increment thime
                        }
                    }
                }
            }

            if(Orangerotten) time++;  // ✅ Time only increases if something rotted this round
        }

        return freshOranges == 0 ? time : -1;
        // if there are any fresh oranges remaining, we couldn't infect them all, but if not we return the time it took to infect them all
    }
};
```
