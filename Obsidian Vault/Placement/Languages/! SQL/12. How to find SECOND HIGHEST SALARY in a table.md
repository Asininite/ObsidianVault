```sql
SELECT name, salary FROM employee
-- this is the outer query that selects the name and salary of employee

WHERE salary = SELECT MAX(salary) 
FROM employee
-- this is the innermost query used to select the max salary

WHERE salary < (SELECT MAX(salary) 
FROM employee));
-- this is the second query that chooses the max salary that is less than the max salary
```
